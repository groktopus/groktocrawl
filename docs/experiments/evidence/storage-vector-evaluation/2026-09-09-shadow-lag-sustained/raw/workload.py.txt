import concurrent.futures
import hashlib
import json
import re
import statistics
import threading
import time
import urllib.request
from datetime import UTC, datetime

BASE = "http://127.0.0.1:8003"
SEED = "groktocrawl-x-shadow-lag-v1"
WRITES = 240
STEADY_SEARCHES = 120
CONCURRENCY = 4


def http(method, path, payload=None):
    body = None if payload is None else json.dumps(payload, sort_keys=True).encode()
    request = urllib.request.Request(BASE + path, data=body, method=method, headers={"Content-Type": "application/json"})
    started = time.monotonic()
    with urllib.request.urlopen(request, timeout=180) as response:
        raw = response.read()
        status = response.status
    return status, raw, (time.monotonic() - started) * 1000


def metrics_text():
    with urllib.request.urlopen(BASE + "/metrics", timeout=10) as response:
        return response.read().decode()


def metric_values(text, prefix):
    values = {}
    for line in text.splitlines():
        if line.startswith("#") or not line.startswith(prefix):
            continue
        name, raw_value = line.rsplit(" ", 1)
        values[name] = float(raw_value)
    return values


def summarize(values):
    values = sorted(values)
    return {
        "count": len(values),
        "minimum_ms": round(values[0], 3),
        "median_ms": round(statistics.median(values), 3),
        "p95_ms": round(values[max(0, int(len(values) * 0.95 + 0.999) - 1)], 3),
        "maximum_ms": round(values[-1], 3),
    }


samples = []
sampling = True


def sample_metrics():
    while sampling:
        text = metrics_text()
        samples.append({
            "monotonic": time.monotonic(),
            "inflight": metric_values(text, "groktocrawl_pgvector_shadow_inflight"),
        })
        time.sleep(0.1)


def write_and_search(index):
    token = hashlib.sha256(f"{SEED}:{index}".encode()).hexdigest()[:16]
    url = f"https://pilot.invalid/sustained/{token}"
    content = f"Synthetic enterprise agent factory evidence {token} with durable orchestration, governance, and retained artifacts."
    index_status, index_body, index_ms = http("POST", "/index", {"url": url, "title": f"Lag sample {index}", "content": content})
    search_status, search_body, search_ms = http("POST", "/search/vector", {"query": content, "limit": 5})
    return {
        "index_status": index_status,
        "search_status": search_status,
        "index_ms": index_ms,
        "search_ms": search_ms,
        "index_digest": hashlib.sha256(index_body).hexdigest(),
        "search_digest": hashlib.sha256(search_body).hexdigest(),
    }


utc_start = datetime.now(UTC).isoformat()
baseline = metrics_text()
thread = threading.Thread(target=sample_metrics, daemon=True)
thread.start()
phase_started = time.monotonic()
with concurrent.futures.ThreadPoolExecutor(max_workers=CONCURRENCY) as pool:
    mixed = list(pool.map(write_and_search, range(WRITES)))
mixed_seconds = time.monotonic() - phase_started

drain_started = time.monotonic()
while True:
    current = metrics_text()
    inflight = metric_values(current, "groktocrawl_pgvector_shadow_inflight")
    if not inflight or all(value <= 0 for value in inflight.values()):
        break
    if time.monotonic() - drain_started > 60:
        raise RuntimeError("shadow backlog did not drain within 60 seconds")
    time.sleep(0.05)
drain_seconds = time.monotonic() - drain_started
after_mixed = metrics_text()


def steady_search(index):
    token = hashlib.sha256(f"{SEED}:{index % WRITES}".encode()).hexdigest()[:16]
    content = f"Synthetic enterprise agent factory evidence {token} with durable orchestration, governance, and retained artifacts."
    status, body, elapsed = http("POST", "/search/vector", {"query": content, "limit": 5})
    return status, hashlib.sha256(body).hexdigest(), elapsed


steady_started = time.monotonic()
with concurrent.futures.ThreadPoolExecutor(max_workers=CONCURRENCY) as pool:
    steady = list(pool.map(steady_search, range(STEADY_SEARCHES)))
steady_seconds = time.monotonic() - steady_started
time.sleep(2)
sampling = False
thread.join(timeout=5)
final = metrics_text()
utc_end = datetime.now(UTC).isoformat()

latency_metrics = metric_values(final, "groktocrawl_pgvector_shadow_completion_lag_seconds")
comparison_metrics = metric_values(final, "groktocrawl_pgvector_shadow_comparisons_total")

print(json.dumps({
    "schema_version": "pgvector-shadow-sustained-lag/1",
    "manifest": {
        "seed": SEED,
        "writes": WRITES,
        "immediate_searches": WRITES,
        "steady_searches": STEADY_SEARCHES,
        "concurrency": CONCURRENCY,
        "shadow_sample_rate": 1.0,
        "stop_conditions": {"backlog_drain_timeout_seconds": 60, "http_timeout_seconds": 180},
        "utc_start": utc_start,
        "utc_end": utc_end,
    },
    "mixed": {
        "duration_seconds": round(mixed_seconds, 3),
        "index_http_failures": sum(item["index_status"] != 201 for item in mixed),
        "search_http_failures": sum(item["search_status"] != 200 for item in mixed),
        "index_latency": summarize([item["index_ms"] for item in mixed]),
        "search_latency": summarize([item["search_ms"] for item in mixed]),
        "response_digest_count": len({item["search_digest"] for item in mixed}),
    },
    "drain_seconds": round(drain_seconds, 6),
    "steady": {
        "duration_seconds": round(steady_seconds, 3),
        "http_failures": sum(item[0] != 200 for item in steady),
        "latency": summarize([item[2] for item in steady]),
        "response_digest_count": len({item[1] for item in steady}),
    },
    "metrics": {
        "baseline_comparisons": metric_values(baseline, "groktocrawl_pgvector_shadow_comparisons_total"),
        "after_mixed_comparisons": metric_values(after_mixed, "groktocrawl_pgvector_shadow_comparisons_total"),
        "final_comparisons": comparison_metrics,
        "completion_lag_histogram": latency_metrics,
        "maximum_sampled_inflight": max((max(sample["inflight"].values(), default=0) for sample in samples), default=0),
        "sample_count": len(samples),
    },
}, indent=2, sort_keys=True))
